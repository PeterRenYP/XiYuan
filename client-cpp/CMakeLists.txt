cmake_minimum_required(VERSION 3.22)
project(ModelObservatory VERSION 1.0.4 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Test Network)
qt_add_executable(ModelObservatory WIN32
    prototypeui.cpp verification_ui.cpp chatprompt.h mcpbridge.cpp mcpbridge.h main.cpp rules.cpp comparison.cpp debugging.cpp chattokenview.cpp chattokenview.h weightfilewindow.cpp weightfilewindow.h project.cpp conversation.cpp features.cpp features.h neurons.cpp neurons.h verification_neurons.cpp uiscale.cpp uiscale.h transport.cpp transport.h visuals.cpp visuals.h workbench.cpp workbench.h verification.cpp verification_maintenance.cpp verification_plan.cpp verification_sessions.cpp anomalycontroller.cpp anomalycontroller.h anomalyengine.cpp anomalyengine.h)
target_link_libraries(ModelObservatory PRIVATE Qt6::Widgets Qt6::Test Qt6::Network)
target_sources(ModelObservatory PRIVATE version.rc)
set_target_properties(ModelObservatory PROPERTIES OUTPUT_NAME Xiyuan)
qt_add_resources(ModelObservatory ui_resources PREFIX "/ui" FILES chevron-down.xpm checkmark.xpm assets/xiyuan.ico)
target_compile_definitions(ModelObservatory PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
if(MSVC)
    target_compile_options(ModelObservatory PRIVATE /utf-8 /W4)
else()
    target_compile_options(ModelObservatory PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS ModelObservatory RUNTIME DESTINATION .)

# 独立检测进程只链接基础与网络库，不加载推理运行时。
qt_add_executable(XiyuanMonitor anomalymain.cpp anomalyengine.cpp anomalyengine.h)
target_link_libraries(XiyuanMonitor PRIVATE Qt6::Core Qt6::Network)
set_target_properties(XiyuanMonitor PROPERTIES OUTPUT_NAME Xiyuan-Monitor)
target_compile_definitions(XiyuanMonitor PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
if(MSVC)
    target_compile_options(XiyuanMonitor PRIVATE /utf-8 /W4)
else()
    target_compile_options(XiyuanMonitor PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS XiyuanMonitor RUNTIME DESTINATION .)

include(CTest)
if(BUILD_TESTING)
    qt_add_executable(AnomalyTests anomalytests.cpp anomalyengine.cpp anomalycontroller.cpp transport.cpp)
    target_link_libraries(AnomalyTests PRIVATE Qt6::Widgets Qt6::Test Qt6::Network)
    target_compile_definitions(AnomalyTests PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
    if(MSVC)
        target_compile_options(AnomalyTests PRIVATE /utf-8)
    endif()
    add_test(NAME AnomalyTests COMMAND AnomalyTests -platform offscreen)
endif()
